SELECT orders.order_id, SUM(products.price * line_items.quantity) as total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON products.product_id = line_items.product_id GROUP BY orders.order_id ORDER BY orders.order_id ASC LIMIT 5;
SELECT c.customer_name, AVG(total_price) as average_order_price FROM customers AS c JOIN (SELECT orders.order_id, customer_id, SUM(products.price * line_items.quantity) as total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON products.product_id = line_items.product_id GROUP BY orders.order_id ) as t on t.customer_id = c.customer_id GROUP BY c.customer_id, c.customer_name ORDER BY c.customer_name ;
